# 조회시간 10초 내외


	<!-- 경력증명서보고서 목록조회 2017.07.31 이상훈, 경력항목 10개이상 가져오게끔 수정.-->
	<select id="adm013R1" parameterType="map" resultType="map">
		SELECT
			A.NO
			, A.OPFR_ADDR
			, A.ADDR
			, A.NM
			, A.SEX_DIVCD
			, A.BIRD
			, A.RRNO AS DEC_RRNO
			, A.DRV_LCNS_NO
			, WKPL
			, WO_CARNO
			, SDT
			, EDT
			, DECODE(SDT,NULL, NULL, '( '||YR||'년 '||MT||'개월 '|| DT ||'일 )') AS WKPRD
			, TRUNC(PRDSUM/12, 0) ||'년 '||TRUNC(MOD(PRDSUM/12, 1)*12, 0)||'개월 '||ROUND(MOD(MOD(PRDSUM/12, 1)*12, 1)*(365/12), 0)||'일' AS WKPRDSUM
		FROM (
			SELECT 
				NVL(A.NO, B.NO)                                  AS NO
				, DECODE(A.NO, NULL, ''           , A.OPFR_ADDR  ) AS OPFR_ADDR
				, DECODE(A.NO, NULL, B.ADDR       , A.ADDR       ) AS ADDR
				, DECODE(A.NO, NULL, B.NM         , A.NM         ) AS NM
				, DECODE(A.NO, NULL, B.SEX_DIVCD  , A.SEX_DIVCD  ) AS SEX_DIVCD
				, DECODE(A.NO, NULL, B.BIRD       , A.BIRD       ) AS BIRD
				, DECODE(A.NO, NULL, B.RRNO       , A.RRNO       ) AS RRNO
				, DECODE(A.NO, NULL, B.DRV_LCNS_NO, A.DRV_LCNS_NO) AS DRV_LCNS_NO
			FROM (
				SELECT 
					ASSOMBR_NO AS NO
					, NVL(OPFR_LONMAD,OPFR_PNSQ_ADDR) ||' '|| OPFR_DTL_ADDR AS OPFR_ADDR
					, NVL(LONMAD,PNSQ_ADDR) ||' '|| DTL_ADDR AS ADDR
					, ASSOMBR_NM AS NM
					, DECODE(SEX_DIVCD,1,'남',2,'여') AS SEX_DIVCD
					, BIRD
					, SCP.DEC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', ENC_RRNO) AS RRNO
					, DRV_LCNS_NO
				FROM TB_TAAM_ASSOMBR
				WHERE
					ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
					AND WLFR_JN_DT = (
						SELECT
							MAX(WLFR_JN_DT)
						FROM TB_TAAM_ASSOMBR
						WHERE
							ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
					)
			) A
			FULL OUTER JOIN
			(
				SELECT 
					DEPUTY_DRV_PRSN_NO AS NO
					, NVL(LONMAD,PNSQ_ADDR)|| DTL_ADDR  AS ADDR
					, DEPUTY_DRV_PRSN_NM AS NM
					, DECODE(SUBSTR(ENC_RRNO,7,1),1,'남',2,'여') AS SEX_DIVCD
					, 
					CASE 
						WHEN SUBSTR(ENC_RRNO, 7, 1) IN ('1', '2')
						THEN '19' || SUBSTR(ENC_RRNO,1,6)
						ELSE '20' || SUBSTR(ENC_RRNO,1,6)
					END AS BIRD
					, SCP.DEC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', ENC_RRNO) AS RRNO
					, DRV_LCNS_NO
				FROM TB_TAAM_DEPUTYDRV
				WHERE 
					ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
			) B ON A.RRNO = B.RRNO
		) A
		, (
			SELECT
				WKPL
				, WO_CARNO
				, SDT
				, EDT
				, PRD
				, YR
				, MT
				, DT
			FROM (
				SELECT 
					'개인택시' AS WKPL
					, (
						SELECT 
							DISTINCT WO_CARNO
						FROM TB_TAAI_TAXI
						WHERE 
							ASSOMBR_NO = A.ASSOMBR_NO
							AND SNO = (SELECT MAX(SNO) FROM TB_TAAI_TAXI WHERE ASSOMBR_NO = A.ASSOMBR_NO)
					) AS WO_CARNO
					, SDT
					, NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')) AS EDT
					, MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT) AS PRD
					, TRUNC(MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT)/12,0) AS YR
					, TRUNC(MOD(MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT)/12,1)*12,0) AS MT
					, ROUND(MOD(MOD(MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT)/12,1)*12,1)*(365/12),0) AS DT
			FROM TB_TAAM_ASSOMBR A, TB_TAAH_ASSOMBRGOVR B
			WHERE
				A.ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
				AND B.GOVR_MSR_CD IN ('04','05')
				AND A.ASSOMBR_NO = B.ASSOMBR_NO
			)
			UNION ALL
			(
				SELECT 
					'개인.대리' AS WKPL
					, (
						SELECT
							WO_CARNO
						FROM TB_TAAI_TAXI
						WHERE
							TAXI_REG_NO = D.TAXI_REG_NO
							AND CHDT = (SELECT MAX(CHDT) FROM TB_TAAI_TAXI WHERE TAXI_REG_NO = D.TAXI_REG_NO)
					) AS WO_CARNO
					, SDT
					, NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')) AS EDT
					, MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT) AS PRD
					, TRUNC(MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT)/12,0) AS YR
					, TRUNC(MOD(MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT)/12,1)*12,0) AS MT
					, ROUND(MOD(MOD(MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT)/12,1)*12,1)*(365/12),0) AS DT
				FROM TB_TAAM_DEPUTYDRV C, TB_TAAH_DEPUTYDRV D
				WHERE
					C.ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
					AND C.DEPUTY_DRV_PRSN_NO  = D.DEPUTY_DRV_PRSN_NO
			) ORDER BY SDT DESC
		) B,
		(
			SELECT
				SUM(PRD) AS PRDSUM
			FROM (
				SELECT
					MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT) AS PRD
				FROM TB_TAAM_ASSOMBR A, TB_TAAH_ASSOMBRGOVR B
				WHERE 
					A.ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
					AND B.GOVR_MSR_CD IN ('04','05')
					AND A.ASSOMBR_NO = B.ASSOMBR_NO
				UNION ALL
				SELECT
					MONTHS_BETWEEN(NVL(EDT,TO_CHAR(SYSDATE,'YYYYMMDD')), SDT) AS PRD
				FROM TB_TAAM_DEPUTYDRV C, TB_TAAH_DEPUTYDRV D
				WHERE
					C.ENC_RRNO = SCP.ENC_B64_TEST('DAMO', 'SCP', 'PATTERN7.SCPS', #{rrno})
					AND C.DEPUTY_DRV_PRSN_NO  = D.DEPUTY_DRV_PRSN_NO
			)
		) C
		ORDER BY SDT DESC
	</select>
